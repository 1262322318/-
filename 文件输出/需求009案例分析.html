SQL代码示例（表一）

SELECT company AS 所属公司, COUNT(DISTINCT CASE WHEN design_current IN ('已完成', '变更已实施') THEN name END) AS 已实施数量, COUNT(DISTINCT CASE WHEN design_current != '草稿' THEN name END) AS 总数量_除草稿, ROUND( COUNT(DISTINCT CASE WHEN design_current IN ('已完成', '变更已实施') THEN name END) * 1.0 / NULLIF(COUNT(DISTINCT CASE WHEN design_current != '草稿' THEN name END), 0), 4) AS 已实施率 FROM ads.ads_ipd_irs_design_change_kccl_dd WHERE LEFT(approvedTIme, 7) >= '${start_month}' AND LEFT(approvedTIme, 7) <= '${end_month}' AND (company IN (${company}) OR '${company}' = '%') ... GROUP BY company ORDER BY company;
